public abstract class BankAccount {
	
	int accountNumber;
	String accountHolderName;
	double balance;
	
	//No-Arg Constructor
	public BankAccount() {
		
	}

	public BankAccount(int accountNumber, String accountHolderName, double balance) {
		super();
		this.accountNumber = accountNumber;
		this.accountHolderName = accountHolderName;
		this.balance = balance;
	}
	
	
	abstract void calculateInterest();
	
	void displayDetails() {
		System.out.println("Account number: " + accountNumber);
		System.out.println("Account Holder Name: " + accountHolderName);
		System.out.println("Balance: " + balance);
	}
}

public class SavingAccount extends BankAccount {
		
	public SavingAccount(int accountNumber, String accountHolderName, double balance) {
		super(accountNumber, accountHolderName, balance);
	}

	@Override
	void calculateInterest() {
		double interest = balance * 4/100;
		System.out.println("Interest: " + interest +" rate.");
	}
	
	void deposite(double amount) {
		balance = balance + amount;
		System.out.println("Deposited: " + amount);
		System.out.println("Update balance: " + balance);
	}
	
	void withdraw(double amount) {
		if(amount <= balance) {
			balance = balance - amount;
			System.out.println("Withdraw: " + amount);
			System.out.println("Update after withdraw: " + balance);
		}else {
			System.out.println("Insufficient balance..");
		}
	}
}

public class FixedDepositeAccount extends BankAccount {
	
	
	public FixedDepositeAccount(int accountNumber, String accountHolderName, double balance) {
		super(accountNumber, accountHolderName, balance);
	}

	@Override
	void calculateInterest() {
		double interest = balance * 6 / 100;
		System.out.println("Interest: " + interest+" rate.");
	}
	
	void deposite(double amount) {
		balance = balance + amount;
		System.out.println("Deposited: " + amount);
	}
	
	void withdraw(double amount) {
		if(amount <= balance) {
			balance = balance - amount;
			System.out.println("Withdraw: " + amount);
		}else {
			System.out.println("Insufficient balance..");
		}
	}
}

public class BankAccountMain {

	public static void main(String[] args) {

public interface LibraryItem {

	double calculateFine(int daysOverdue);
	void displayItems();
}

public class Book implements LibraryItem {
	String title;
	String author;
	
	public Book(String title, String author) {
		super();
		this.title = title;
		this.author = author;
	}

	@Override
	public double calculateFine(int daysOverdue) {
		return daysOverdue * 2;
	}

	@Override
	public void displayItems() {
		System.out.println("Item type: Book");
		System.out.println("Title: " + title);
		System.out.println("Author: "+ author);
	}
	
	void readSample() {
		System.out.println("Reading book sample...");
	}
}

public class DVD implements LibraryItem {
	String title;
	int durationMinutes;
	
	
	public DVD(String title, int durationMinutes) {
		super();
		this.title = title;
		this.durationMinutes = durationMinutes;
	}

	@Override
	public double calculateFine(int daysOverdue) {
		return daysOverdue * 5;
	}

	@Override
	public void displayItems() {
		System.out.println("Item type: DVD");
		System.out.println("Title: " + title);
		System.out.println("Duration: " + durationMinutes + " minutes");
	}
	
	void playTrailer() {
		System.out.println("Playing DVD Trailer......");
	}
}

public class LibraryItemMain {

	public static void main(String[] args) {
		Book b = new Book("Start with Why", "Simon Sinek");
		
		System.out.println("-------BOOK----------");
		b.displayItems();
		System.out.println("Fine: Rs." + b.calculateFine(5));
		b.readSample();
		
		System.out.println("------==-------=---------------------");
		
		DVD d = new DVD("Avenger", 180);
		d.displayItems();
		System.out.println("Fine: Rs." + d.calculateFine(5));
		d.playTrailer();
		
	}

public interface Payment {
	void processPayment(double amount);
	
	void generateReceipt();
}

interface Refundable{
	void processRefund(double amount);
}

interface Rewardable{
	void addRewardPoints(int points);
}

package Assignment5;

public class CreditCardPayment implements Payment, Refundable, Rewardable {

	@Override
	public void addRewardPoints(int points) {
		System.out.println("Reward Points: " + points); 
	}

	@Override
	public void processPayment(double amount) {
		System.out.println("Credit Card Payment: Rs." + amount);
	}
	
	@Override
	public void processRefund(double amount) {
		double fee = amount * 2 / 100; double refund = amount - fee; 
		System.out.println("Processing Fee: Rs." + fee); 
		System.out.println("Refund Amount: Rs." + refund);
	}

	@Override
	public void generateReceipt() {
		System.out.println("Credit Card Receipt Generated");

		}
	}

public abstract class Vehical {
	String vehicleName;
	int param;
	
	public Vehical(String vehicleName, int param) {
		this.vehicleName = vehicleName;
		this.param = param;
	}
	
	abstract double calculateToll(int param);
	
	void displayDetails() {
		System.out.println("Vehical Name: " + vehicleName);
		System.out.println("Parameter: " + param);
		System.out.println("Toll: Rs." + calculateToll(param));
	}
	
	
}

package Assign5;

public class Car extends Vehical {

	public Car(String vehicleName, int seats) {
		super(vehicleName, seats);
	}

	@Override
	double calculateToll(int param) {
		return param * 20;
	}

}
